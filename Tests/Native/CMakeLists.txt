cmake_minimum_required(VERSION 3.20)
project(UBotMuJoCoNativeTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(PLUGIN "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SDK "${PLUGIN}/ThirdParty/MuJoCo/4.0.0")
set(NATIVE "${PLUGIN}/Source/UBotMuJoCoNative")
add_executable(ubot_mujoco_native_tests test_native.cpp "${NATIVE}/Private/UBotMuJoCoNative.cpp" "${NATIVE}/Private/ThirdParty/tinyxml2.cpp")
target_compile_definitions(ubot_mujoco_native_tests PRIVATE UBOT_MUJOCO_STANDALONE=1)
target_include_directories(ubot_mujoco_native_tests PRIVATE "${NATIVE}/Public" "${SDK}/include")
target_link_libraries(ubot_mujoco_native_tests PRIVATE "${SDK}/lib/Win64/mujoco.lib")
if(MSVC)
  target_compile_options(ubot_mujoco_native_tests PRIVATE /EHsc /W4 /utf-8)
endif()
add_custom_command(TARGET ubot_mujoco_native_tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${SDK}/lib/Win64/mujoco.dll" "$<TARGET_FILE_DIR:ubot_mujoco_native_tests>")
enable_testing()
add_test(NAME native_reference COMMAND ubot_mujoco_native_tests "${CMAKE_CURRENT_SOURCE_DIR}/fixtures")
